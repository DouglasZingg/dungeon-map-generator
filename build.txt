@echo off
title Dungeon Map Generator Builder

cd /d "%~dp0"

echo.
echo ===============================
echo Building Dungeon Generator
echo ===============================
echo.

if exist build rmdir /s /q build
if exist dist rmdir /s /q dist

mkdir build
mkdir dist

echo Compiling...

"C:\Program Files\Java\jdk-21.0.11\bin\javac.exe" ^
    -d build ^
    src\*.java

if errorlevel 1 (
    echo.
    echo Build Failed!
    pause
    exit /b
)

echo.
echo Creating JAR...

cd build

"C:\Program Files\Java\jdk-21.0.11\bin\jar.exe" ^
    cfe DungeonGenerator.jar Main *.class

cd ..

echo.
echo Building EXE...

"C:\Program Files\Java\jdk-21.0.11\bin\jpackage.exe" ^
    --name "Dungeon Map Generator" ^
    --input build ^
    --main-jar DungeonGenerator.jar ^
    --main-class Main ^
    --icon assets\icon.ico ^
    --dest dist ^
    --type exe

echo.
echo ===============================
echo Build Successful!
echo ===============================
echo.

pause
